\chapter*{Source notes}
\begin{itemize}
\item Dalla-Torre et al., \emph{Nucleotide Transformer: building and evaluating robust foundation models for human genomics}, Nature Methods 22 (2025): pretraining/fine-tuning across curated genomic tasks, with strong supervised baselines and task-dependent performance.
\item Nguyen et al. (2023), \emph{HyenaDNA}: long-range single-nucleotide genomic sequence modeling using Hyena operators.
\item Schiff et al. (2024), \emph{Caduceus}: bidirectional long-range DNA modeling with an explicit reverse-complement equivariant construction.
\item 2025 Nature Communications benchmark of DNA foundation models: performance varies by task; specialized models can outperform general foundation embeddings on some applications.
\end{itemize}
